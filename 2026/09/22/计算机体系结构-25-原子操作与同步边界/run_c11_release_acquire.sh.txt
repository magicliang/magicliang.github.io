#!/usr/bin/env sh
set -eu
cd "$(dirname "$0")/.."
cc=${CC:-clang}
out_dir=outputs
mkdir -p "$out_dir"
bin="${TMPDIR:-/tmp}/ca_c11_release_acquire.$$"
trap 'rm -f "$bin"' EXIT
"$cc" -std=c11 -O2 -Wall -Wextra -pedantic -pthread src/c11_release_acquire.c -o "$bin"
"$bin" | tee "$out_dir/c11_release_acquire_output.txt"
