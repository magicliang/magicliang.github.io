"""Idealized DRAM-traffic matrix multiplication model (no measured points)."""

from __future__ import annotations

from math import log10
from pathlib import Path


def model(output: Path) -> dict:
    size, tile, bytes_per_element = 256, 32, 4
    flops = 2 * size**3
    cases = []
    for name, a_b_bytes in (("naive", 8 * size**3), ("tiled", 8 * size**3 // tile)):
        bytes_dram = a_b_bytes + 8 * size**2
        intensity = flops / bytes_dram
        cases.append({"name": name, "dram_bytes_model": bytes_dram, "flops": flops, "operational_intensity_flop_per_byte": intensity, "model_bound_gflops": min(128, 32 * intensity), "measured": False})
    def xcoord(value: float) -> float:
        return 85 + 490 * (log10(value) + 1) / 3
    def ycoord(value: float) -> float:
        return 345 - 285 * (log10(value) + 1) / 3.5
    roof = [(xcoord(10**(step / 20 - 1)), ycoord(min(128, 32 * 10**(step / 20 - 1)))) for step in range(61)]
    path = " ".join(("M" if index == 0 else "L") + f"{x:.1f},{y:.1f}" for index, (x, y) in enumerate(roof))
    dots = "\n".join(f'<circle cx="{xcoord(case["operational_intensity_flop_per_byte"]):.1f}" cy="{ycoord(case["model_bound_gflops"]):.1f}" r="5" fill="#c34332"/><text x="{xcoord(case["operational_intensity_flop_per_byte"])+9:.1f}" y="{ycoord(case["model_bound_gflops"])-8:.1f}" font-size="14">{case["name"]} (model)</text>' for case in cases)
    svg = f'''<svg xmlns="http://www.w3.org/2000/svg" width="640" height="415" viewBox="0 0 640 415" role="img" aria-label="Roofline model upper bound; no measured points">
<rect width="640" height="415" fill="white"/><path d="M85 60 V345 H585" stroke="#222" stroke-width="2" fill="none"/>
<path d="{path}" stroke="#246b9c" stroke-width="3" fill="none"/>
{dots}
<text x="220" y="395" font-size="16">DRAM FLOP/byte (log scale)</text>
<text x="6" y="42" font-size="14">GFLOP/s (log)</text>
<text x="330" y="105" font-size="13">128 GFLOP/s compute; 32 GB/s DRAM model</text>
</svg>\n'''
    output.write_text(svg, encoding="utf-8")
    return {"evidence_level": "解析模型，非实测", "matrix": "C=A×B; square row-major float32, no FMA exception", "size": size, "tile_size": tile, "storage_level": "ideal DRAM traffic (no conflict/capacity misses, no write allocate)", "flop_accounting": "one multiply and one add per inner iteration = 2*N^3", "c_bytes": 8 * size**2, "naive_a_b_bytes": 8 * size**3, "tiled_a_b_bytes": 8 * size**3 // tile, "peak_gflops_assumed": 128, "bandwidth_gb_s_assumed": 32, "points": cases, "measured_points": [], "plot": output.name}
